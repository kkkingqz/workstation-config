# apt packages every host needs on top of a standard Ubuntu desktop.
# One package per line; `ppa:OWNER/NAME` adds an Ubuntu PPA first;
# `snap-remove:NAME` removes a snap of the standard install.
# Installed by bootstrap.sh; `ws check apt` compares with apt-mark showmanual.
# Host packages: nix/hosts/<name>/apt.txt.

# Nix: delivers the user layer (helpws history-nix)
nix-bin
nix-setup-systemd

# Login shell (fish 4.x: terminal/fish), terminal, repository
ppa:fish-shell/release-4
fish
ghostty
git
curl

# Python helpers use the host PyGObject and AT-SPI (ws-nautilus-current)
python3-gi
gir1.2-atspi-2.0

# Qt applications under GNOME (rebuild.md 6.2)
qgnomeplatform-qt5
qgnomeplatform-qt6
qtwayland5
qt6-wayland

# Distrobox / Podman (rebuild.md 6.3)
podman
distrobox
uidmap
fuse-overlayfs
slirp4netns
passt

# Flatpak applications (wsflatpak)
flatpak
# udev rules for game controllers of the Steam flatpak
steam-devices

# Btrfs subvolumes, snapshots and recovery (system-backup-snapshot)
btrfs-progs

# sg: bootstrap.sh runs the first switch with nix-users before a new login
# (Ubuntu 26.04 moved sg/newgrp from login to util-linux-extra)
util-linux-extra

# Virtual machines: KVM/QEMU, libvirt qemu:///system, virt-manager (helpws virt).
# Recommends are not installed, so the SPICE console parts are listed.
qemu-system-x86
qemu-utils
qemu-system-modules-spice
qemu-system-modules-opengl
libvirt-daemon-system
libvirt-clients
virt-manager
gir1.2-spiceclientgtk-3.0
gir1.2-spiceclientglib-2.0
# UEFI (with Secure Boot variants) and TPM: a Windows 11 guest needs both
ovmf
swtpm-tools
# Shared folders (virtiofs)
virtiofsd

# Desktop and command-line tools on top of the standard install
# Space preview in Nautilus (org.gnome.NautilusPreviewer)
gnome-sushi
# "This computer" (drives) in Nautilus
nautilus-my-computer
gnome-system-monitor
btop
# GitHub CLI (gh auth for git push)
gh
# MPRIS players and PipeWire from the command line (playerctl, pactl)
playerctl
pulseaudio-utils
# Crash dumps (coredumpctl)
systemd-coredump
# Non-free codecs and Microsoft fonts (multiverse); the font installer asks
# to accept its EULA during bootstrap.sh
ubuntu-restricted-extras
